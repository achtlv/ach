auto main() {
    string raw = q_get("key");
    if (raw == "") { out("missing ?key=\n"); return; }
    raw = replace(raw, "^", " ");
    string chk = regex(raw, "^ *([+-]?[0-9]+([.][0-9]+)?) +([+-]?[0-9]+([.][0-9]+)?) *$");
    if (chk == "") { out("bad format, expect: m^n\n"); return; }
    number m = to_num(regex(raw, "^ *([+-]?[0-9]+([.][0-9]+)?)"));
    number n = to_num(regex(raw, "([+-]?[0-9]+([.][0-9]+)?) *$"));
    out(to_str(m) + "^" + to_str(n) + "=" + to_str(pow(m, n)) + "\n");
    return;
}
