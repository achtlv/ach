auto main() {
    string UA = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36";
    string base = "https://miaogongzi.lanzout.com";
    string share = base + "/b01rgkhhe";

    // 1) 分享页
    string html = get(share)
                  with UA UA
                  with header { "Accept": "text/html" };
    if (status() != 200) {
        return with 502;
    }

    string t    = regex(html, "var ibm0zt = '([^']+)'");
    string k    = regex(html, "var _hkd1m = '([^']+)'");
    string fid  = regex(html, "'fid':([0-9]+)");
    string uid  = regex(html, "'uid':'([^']+)'");
    string puid = regex(html, "'puid':'([^']+)'");
    if (t == "" || k == "" || fid == "") {
        return with 502;
    }

    // 2) 文件列表（第 1 条一般是最新）
    string form =
        "lx=2&fid=" + fid +
        "&uid=" + uid +
        "&pg=1&t=" + t +
        "&k=" + k +
        "&puid=" + puid +
        "&webfoldersign=";

    string list = post(base + "/filemoreajax.php?file=" + fid, form)
                  with UA UA
                  with header {
                      "Content-Type": "application/x-www-form-urlencoded",
                      "Referer": share
                  };

    if (json_num(list, "$.zt") != 1) {
        return with 502;
    }

    string fileId = json_str(list, "$.text[0].id");
    string fname  = json_str(list, "$.text[0].name_all");
    if (fileId == "") {
        return with 404;
    }

    // 可选：按文件名过滤（例如只要带 releaseA 的）
    // string want = q_get("name");
    // if (want != "" && !contains(fname, want)) { ... 遍历 text[i] ... }

    // 3) 文件页 → iframe
    string page = get(base + "/" + fileId)
                  with UA UA
                  with header { "Referer": share };
    string fn = regex(page, "src=\"(/fn\\?[^\"]+)\"");
    if (fn == "") {
        return with 502;
    }

    // 4) iframe → sign / ajax 域名
    string iframe = get(base + fn)
                    with UA UA
                    with header { "Referer": base + "/" + fileId };

    string wp_sign  = regex(iframe, "var wp_sign = '([^']+)'");
    string ajaxdata = regex(iframe, "var ajaxdata = '([^']+)'");
    string domain2  = regex(iframe, "var domain2 = '([^']+)'");
    if (wp_sign == "" || domain2 == "") {
        return with 502;
    }
    if (ajaxdata == "") {
        ajaxdata = "YiQi";
    }

    // 5) 直链接口
    string postBody =
        "action=downprocess&websignkey=" + ajaxdata +
        "&signs=" + ajaxdata +
        "&sign=" + wp_sign +
        "&websign=&kd=0&ves=1";

    string dl = post(domain2, postBody)
                with UA UA
                with header {
                    "Content-Type": "application/x-www-form-urlencoded",
                    "Referer": base + "/"
                };

    string dom = json_str(dl, "$.dom");
    string path = json_str(dl, "$.url");
    if (dom == "" || path == "") {
        return with 502;
    }

    // 6) 跳转真实下载地址（临时直链，有效期短）
    return dom + "/file/" + path with 302;
}
